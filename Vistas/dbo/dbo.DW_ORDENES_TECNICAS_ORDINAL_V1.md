## Definición (CREATE VIEW)
```sql
-- Vista: dbo.DW_ORDENES_TECNICAS_ORDINAL_V1
-- Extraida: 2026-08-07T15:27:32.664577+00:00
-- Fuente: sys.sql_modules / OBJECT_DEFINITION

CREATE VIEW [dbo].[DW_ORDENES_TECNICAS_ORDINAL_V1] AS with centrooperativoN as (
SELECT --top 1 
	CO.CENTROOPERATIVONOMBRE, CO.CENTROOPERATIVOID, ord.EMPRESAID , ORD.ORDENTRBRED
		FROM SIGASC.CENTROOPERATIVO CO 
			INNER JOIN sigasc.empresasmulti em
				on CO.empresaid = em.empresaid
			INNER JOIN SIGASC.CENTROOPERATIVOSUCURSAL CS 
			on		CS.EMPRESAID			= CO.EMPRESAID 
				AND CS.CENTROOPERATIVOID	= CO.CENTROOPERATIVOID 
			INNER JOIN SIGASC.VM_CLIENTE CLI
			on			CO.EMPRESAID = CLI.EMPRESAID 
					AND CS.CENTROOPESUCURSALID = CLI.SUCURSALID  
					AND CS.EMPRESAID = CLI.EMPRESAID
			INNER JOIN SIGASC.ordensrv ord
			on 			CLI.empresaid = ord.empresaid
					and ord.CLIENTENROORD = CLI.clientenro 
), 
ORDENTRB_CORREGIDA as (
	SELECT  CO.EMPRESAID
		,ORDTRBNRO			
		,ORDTRBSTS			
		,ORDTRBFCH			
		,ORDTRBFCHINICIAL		
		,ORDTRBFCHFINAL		
		,ORDTRBHORAINICIAL	
		,ORDTRBHORAFINAL		
		,ORDTRBCPTOID			
		,ORDTRBCORTE
		,TECNICOID
		,TECNICO2ID
		,MOTIVOORDID
		, case when ORDTRBFCHFINAL is null then NULL
			WHEN ORDTRBFCHFINAL IS NOT NULL AND  ORDTRBHORAFINAL IS NULL THEN 
				DATEFROMPARTS(year(ORDTRBFCHFINAL), MONTH(ORDTRBFCHFINAL), day(ORDTRBFCHFINAL))
			WHEN ORDTRBFCHFINAL IS NOT NULL AND  ORDTRBHORAFINAL IS NOT NULL THEN 
				DATETIMEFROMPARTS(year(ORDTRBFCHFINAL), MONTH(ORDTRBFCHFINAL), day(ORDTRBFCHFINAL), DATEPART(hour, ORDTRBHORAFINAL), DATEPART(minute,ORDTRBHORAFINAL), DATEPART(second,ORDTRBHORAFINAL),0)
			else 
				NULL
			end ORDTRBFECHA_FINAL
		, case when ORDTRBFCHINICIAL is null then NULL
			WHEN ORDTRBFCHINICIAL IS NOT NULL AND  ORDTRBHORAINICIAL IS NULL THEN 
				DATEFROMPARTS(year(ORDTRBFCHINICIAL), MONTH(ORDTRBFCHINICIAL), day(ORDTRBFCHINICIAL))
			WHEN ORDTRBFCHINICIAL IS NOT NULL AND  ORDTRBHORAINICIAL IS NOT NULL THEN 
				DATETIMEFROMPARTS(year(ORDTRBFCHINICIAL), MONTH(ORDTRBFCHINICIAL), day(ORDTRBFCHINICIAL), DATEPART(hour, ORDTRBHORAINICIAL), DATEPART(minute,ORDTRBHORAINICIAL), DATEPART(second,ORDTRBHORAINICIAL),0)
			else 
				NULL
			end ORDTRBFECHA_INICIAL
		FROM SIGASC.ORDENTRB CO
			INNER JOIN sigasc.empresasmulti em
				on CO.empresaid = em.empresaid
), 
datos as (
	SELECT
		R0.EMPRESAID,
		R0.ORDTRBNRO			as ORDINAL_NRO,
		R0.ORDTRBSTS			AS ORDINAL_STS,
		R0.ORDTRBFCH			AS ORDINAL_F_CREACION,
		R0.ORDTRBFCHINICIAL		AS ORDINAL_F_INI,
		R0.ORDTRBFCHFINAL		AS ORDINAL_F_FIN,
		R0.ORDTRBHORAINICIAL	ORDINAL_H_INI,
		R0.ORDTRBHORAFINAL		AS ORDINAL_H_FIN,
		R0.ORDTRBCPTOID			AS ORDINAL_CONCEPTO,
		R0.ORDTRBCORTE			AS ZONA_CORTADA,
		DATEDIFF(HOUR, ORDTRBFECHA_INICIAL, ORDTRBFECHA_FINAL) "DIFERENCIA_EN_HORAS",
		DATEDIFF(HOUR, ORDTRBFECHA_INICIAL, ORDTRBFECHA_FINAL)/24 "DIFERENCIA_EN_HORAS_24",
		cop.CENTROOPERATIVONOMBRE	"CO",
		cop.CENTROOPERATIVOID		"COperativoID",
		R0.TECNICOID				AS ID_TECNICO_PRIN,
		TRIM(TP.TECNICONOMBRE)		AS NOMBRE_TECNICO_PRIN,
		TP.TECNICOEMPLEADONRO		AS LEGAJO_TEC_PRIN,
		R0.TECNICO2ID				AS ID_TECNICO_AUX,
		TRIM(TA.TECNICONOMBRE)		AS NOMBRE_TECNICO_AUX,
		TA.TECNICOEMPLEADONRO		AS LEGJO_TEC_AUX,
		TRIM(NCO.ORDTRBCPTONOMBRE)	ORDINAL_CONCEPTO_NOMBRE, 
		TRIM(MO.MOTIVOORDNOMBRE)	AS  MOTIVO_CUMPLIMIENTO,
		R0.MOTIVOORDID				IDMOTIVO_CUMPLIMIENTO, 
		CASE WHEN R11.TOTAL IS NULL THEN 0 ELSE R11.TOTAL END AS TOTAL_ORDENES,
		R0.ORDTRBFCH,
		r0.ORDTRBNRO
		FROM ORDENTRB_CORREGIDA R0 --- falta
			left join centrooperativoN cop 
				on cop.EMPRESAID = R0.EMPRESAID
					and cop.ORDENTRBRED =  R0.ORDTRBNRO
			LEFT join SIGASC.MOTIVOORD MO 
				ON R0.MOTIVOORDID = MO.MOTIVOORDID
			LEFT join SIGASC.CPTOORDTRB NCO 
				ON NCO.EMPRESAID = R0.EMPRESAID 
					AND NCO.ORDTRBCPTOID = R0.ORDTRBCPTOID
			LEFT JOIN SIGASC.TECNICO TP 
				ON TP.EMPRESAID = R0.EMPRESAID 
					AND TP.TECNICOID = R0.TECNICOID
			LEFT JOIN SIGASC.TECNICO TA 
				ON TA.EMPRESAID = R0.EMPRESAID 
					AND TA.TECNICOID = R0.TECNICO2ID
			LEFT JOIN (SELECT ORD.EMPRESAID, ORD.ORDENTRBRED, COUNT(*) AS TOTAL
   						FROM SIGASC.ORDENSRV ORD
            			WHERE ORD.ORDENTPO = 'R'
            			GROUP BY ORD.EMPRESAID, ORD.ORDENTRBRED) R11 
	   			ON R11.EMPRESAID = R0.EMPRESAID 
   					AND R11.ORDENTRBRED = R0.ORDTRBNRO 
)
Select * from datos;
```
