## Definición (CREATE VIEW)
```sql
-- Vista: dbo.v_EscalonPromo
-- Extraida: 2026-08-07T15:27:52.731481+00:00
-- Fuente: sys.sql_modules / OBJECT_DEFINITION

CREATE VIEW [dbo].[v_EscalonPromo]
AS SELECT EMPRESAID,PROMOCIONID,PROMOCIONMES
FROM (
  SELECT 
    pm.EMPRESAID,pm.PROMOCIONID,pm.PROMOCIONMES,PROMOCIONPRC,
    LAG(pm.PROMOCIONPRC) OVER (PARTITION BY pm.EMPRESAID,pm.PROMOCIONID ORDER BY pm.PROMOCIONMES) AS prev_PROMOCIONPRC
  FROM SIGASC.PROMOCIONMES pm
LEFT JOIN SIGASC.PROMOCION p
ON p.PROMOCIONID = pm.PROMOCIONID
and p.EMPRESAID = pm.EMPRESAID
where PROMOCIONTPODTO = 'F'
and PROMOCIONMES not in (0,1)
) t
WHERE (PROMOCIONPRC != prev_PROMOCIONPRC OR prev_PROMOCIONPRC IS NULL)
and PROMOCIONMES <>2
UNION ALL
SELECT EMPRESAID,PROMOCIONID,PROMOCIONMES
FROM (
  SELECT 
    pm.EMPRESAID,pm.PROMOCIONID,pm.PROMOCIONMES,PROMOCIONDTOPRJ,
    LAG(pm.PROMOCIONDTOPRJ) OVER (PARTITION BY pm.EMPRESAID,pm.PROMOCIONID ORDER BY pm.PROMOCIONMES) AS prev_PROMOCIONDTOPRJ
  FROM SIGASC.PROMOCIONMES pm
LEFT JOIN SIGASC.PROMOCION p
ON p.PROMOCIONID = pm.PROMOCIONID
and p.EMPRESAID = pm.EMPRESAID
where PROMOCIONTPODTO = 'P'
and PROMOCIONMES not in (0,1)
) t
WHERE (PROMOCIONDTOPRJ != prev_PROMOCIONDTOPRJ OR prev_PROMOCIONDTOPRJ IS NULL)
and PROMOCIONMES <>2
UNION ALL
SELECT EMPRESAID,PROMOCIONID,PROMOCIONMES
FROM (
  SELECT 
    pm.EMPRESAID,pm.PROMOCIONID,pm.PROMOCIONMES,PROMOCIONDTOPRC,
    LAG(pm.PROMOCIONDTOPRC) OVER (PARTITION BY pm.EMPRESAID,pm.PROMOCIONID ORDER BY pm.PROMOCIONMES) AS prev_PROMOCIONDTOPRC
  FROM SIGASC.PROMOCIONMES pm
LEFT JOIN SIGASC.PROMOCION p
ON p.PROMOCIONID = pm.PROMOCIONID
and p.EMPRESAID = pm.EMPRESAID
where PROMOCIONTPODTO = 'I'
and PROMOCIONMES not in (0,1)
) t
WHERE (PROMOCIONDTOPRC != prev_PROMOCIONDTOPRC OR prev_PROMOCIONDTOPRC IS NULL)
and PROMOCIONMES <>2;
```
